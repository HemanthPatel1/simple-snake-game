# Largest Number Code Logic

## Problem
Find the largest number from a list of numbers.

## Approach 1: Using a Loop

def find_largest(numbers):
    if not numbers:
        return None  # Return None if the list is empty

    largest = numbers[0]  # Assume the first element is the largest

    for num in numbers:
        if num > largest:
            largest = num  # Update largest if current number is greater

    return largest

# Example usage:
numbers = [3, 7, 1, 9, 4, 6, 2, 8, 5]
result = find_largest(numbers)
print(f"The largest number is: {result}")  # Output: The largest number is: 9


## Approach 2: Using Python's Built-in max()

def find_largest_builtin(numbers):
    if not numbers:
        return None
    return max(numbers)

# Example usage:
numbers = [15, 42, 8, 23, 4, 100, 57]
result = find_largest_builtin(numbers)
print(f"The largest number is: {result}")  # Output: The largest number is: 100


## Approach 3: Using Sorting

def find_largest_sorting(numbers):
    if not numbers:
        return None
    sorted_numbers = sorted(numbers, reverse=True)
    return sorted_numbers[0]  # The first element after descending sort is the largest

# Example usage:
numbers = [34, 12, 89, 56, 78, 23]
result = find_largest_sorting(numbers)
print(f"The largest number is: {result}")  # Output: The largest number is: 89


## Approach 4: Comparing Three Numbers (Simple Case)

def find_largest_of_three(a, b, c):
    if a >= b and a >= c:
        return a
    elif b >= a and b >= c:
        return b
    else:
        return c

# Example usage:
a, b, c = 45, 78, 32
result = find_largest_of_three(a, b, c)
print(f"The largest of {a}, {b}, {c} is: {result}")  # Output: The largest of 45, 78, 32 is: 78


## JavaScript Version

function findLargest(numbers) {
    if (numbers.length === 0) return null;

    let largest = numbers[0];

    for (let i = 1; i < numbers.length; i++) {
        if (numbers[i] > largest) {
            largest = numbers[i];
        }
    }

    return largest;
}

// Using Math.max with spread operator
function findLargestBuiltin(numbers) {
    if (numbers.length === 0) return null;
    return Math.max(...numbers);
}

// Example usage:
const nums = [5, 3, 8, 1, 9, 2, 7, 4, 6];
console.log("Largest:", findLargest(nums));        // Output: Largest: 9
console.log("Largest:", findLargestBuiltin(nums)); // Output: Largest: 9


## Time and Space Complexity

- Loop approach:    Time O(n), Space O(1)  — optimal
- Built-in max():   Time O(n), Space O(1)  — optimal
- Sorting approach: Time O(n log n), Space O(n) — less efficient
- Three-number:     Time O(1), Space O(1)  — constant (fixed input size)
